query matchResults($startDate: String, $endDate: String, $startIndex: Int, $endIndex: Int, $teamId: String) {
  timeOffset {
    fb
  }
  matchNumByDate(startDate: $startDate, endDate: $endDate, teamId: $teamId) {
    total
  }
  matches: matchResult(startDate: $startDate, endDate: $endDate, startIndex: $startIndex,endIndex: $endIndex, teamId: $teamId) {
    id
    status
    frontEndId
    matchDayOfWeek
    matchNumber
    matchDate
    kickOffTime
    sequence
    homeTeam {
      id
      name_en
      name_ch
    }
    awayTeam {
      id
      name_en
      name_ch
    }
    tournament {
      code
      name_en
      name_ch
    }
    results {
      homeResult
      awayResult
      resultConfirmType
      payoutConfirmed
      stageId
      resultType
      sequence
    }
    poolInfo {
      payoutRefundPools
      refundPools
      ntsInfo
      entInfo
      definedPools
    }
  }
}
